Standalone Clean-Room Parser Pseudocode

function ReadEdFileTypeList(buf, tl_off):
    cnt = ReadU32LE(buf, tl_off + 9); p = tl_off + 13; types = []
    for i in 0 .. cnt - 1:
        if buf[p:p+5] != b"Type\0": break
        p += 5; nlen = ReadU32LE(buf, p); p += 4
        nm = buf[p:p+nlen].rstrip(b"\0").decode("ascii"); p += nlen
        sz = ReadI32LE(buf, p); p += 4; types.append((nm, sz))
    return types
