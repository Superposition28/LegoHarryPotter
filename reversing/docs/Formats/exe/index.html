Standalone Clean-Room Parser Pseudocode

function IdentifyPeTarget(pe_bytes):
    lfanew = ReadU32LE(pe_bytes, 0x3C)
    if pe_bytes[lfanew : lfanew+4] != b"PE\0\0": return "Invalid"
    machine = ReadU16LE(pe_bytes, lfanew + 4)
    return "PE32+ AMD64" if machine == 0x8664 else "PE32 x86"
